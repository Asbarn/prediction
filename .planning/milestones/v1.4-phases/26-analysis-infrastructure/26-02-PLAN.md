2. Create src/analysis/output.rs with:

   - `OutputFormat` enum with variants `Table` and `Json`. Derive `Debug, Clone, clap::ValueEnum`. This allows clap to parse `--output table` and `--output json` automatically.

   - `render_output<T: Serialize>(data: &T, format: &OutputFormat, table_fn: impl FnOnce(&T) -> Table)`:
     * On OutputFormat::Table: call table_fn(data) to get a comfy_table::Table, then println!("{table}").
     * On OutputFormat::Json: call serde_json::to_string_pretty(data).unwrap() and println.

   - `new_table(headers: &[&str]) -> Table`:
     * Create a new comfy_table::Table.
     * Set ContentArrangement::Dynamic.
     * Set headers from the provided slice.
     * Return the table (callers set column alignment per their needs).

   - `set_numeric_columns(table: &mut Table, columns: &[usize])`:
     * For each column index, call `table.column_mut(idx)` and set `CellAlignment::Right`.
     * This is a helper so CLI binaries don't need to import comfy_table types directly.

   - `section_header(table: &mut Table, text: &str, col_count: usize)`:
     * Add a row where the first cell contains the text and remaining cells are empty strings.
     * This creates visual section breaks in the table output.

   - `LoadingSummary` struct with `#[derive(Serialize)]`: fields `date_range: String`, `files_loaded: usize`, `files_missing: usize`, `records_loaded: usize`, `parse_errors: usize`, and optionally `events_found: usize`.

   - `render_loading_summary(summary: &LoadingSummary, format: &OutputFormat)`:
     * For Table: create a 2-column table ("Metric", "Value") with right-justified Value column, add rows for each field.
     * For Json: serialize to pretty JSON.
     * This is the placeholder output that Phase 26 delivers before Phases 27-28 add analysis sections.

   - Re-export comfy_table::Table from output.rs so downstream code can use `prediction::analysis::output::Table` without adding comfy-table as a direct dependency.

2. Create src/bin/spread_analytics.rs:

   ```rust
   use anyhow::Result;
   use clap::Parser;
   use chrono::NaiveDate;
   use std::path::PathBuf;
   use prediction::analysis::io::{DateRange, load_jsonl};
   use prediction::analysis::output::{OutputFormat, LoadingSummary, render_loading_summary};
   use prediction::spread::patterns::SpreadResult;

   #[derive(Parser)]
   #[command(name = "spread-analytics")]
   #[command(about = "Analyze spread distribution patterns from recorded JSONL data")]
   struct Cli {
       /// Start date (YYYY-MM-DD)
       #[arg(long)]
       from: Option<NaiveDate>,

       /// End date (YYYY-MM-DD), defaults to today
       #[arg(long)]
       to: Option<NaiveDate>,

       /// Analyze last N days (alternative to --from/--to)
       #[arg(long)]
       last: Option<u32>,

       /// Output format: table (default) or json
       #[arg(long, default_value = "table")]
       output: OutputFormat,

       /// Break down results by event_id
       #[arg(long)]
       by_event: bool,

       /// Spread logs directory
       #[arg(long, default_value = "spread_logs")]
       log_dir: PathBuf,
   }

   fn main() -> Result<()> {
       let cli = Cli::parse();
       let range = DateRange::from_args(cli.from, cli.to, cli.last)?;
       let files = range.files_in_dir(&cli.log_dir);

       let result = load_jsonl::<SpreadResult>(&files);

       if result.errors > 0 {
           eprintln!("Warning: {} malformed JSONL lines skipped", result.errors);
       }

       // Count unique event_ids if --by-event is set
       let events_found = if cli.by_event {
           let mut event_ids: std::collections::HashSet<&str> = std::collections::HashSet::new();
           for record in &result.records {
               event_ids.insert(&record.event_id);
           }
           event_ids.len()
       } else {
           0
       };

       let summary = LoadingSummary {
           date_range: range.to_string(),
           files_loaded: result.files_loaded,
           files_missing: result.files_missing,
           records_loaded: result.records.len(),
           parse_errors: result.errors,
           events_found,
       };

       // Phase 27 will replace this with actual spread analysis rendering
       render_loading_summary(&summary, &cli.output);

       if cli.by_event && events_found > 0 {
           eprintln!("Found {} unique events (per-event analysis added in Phase 27)", events_found);
       }

       Ok(())
   }
   ```
