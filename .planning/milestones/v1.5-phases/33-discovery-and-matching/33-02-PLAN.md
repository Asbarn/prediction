**3. Add Derive absence checking in step 1b:**

After the Polymarket absence checking block (around line 602-618), add a Derive block:
```rust
if let Some(ref derive_inst) = mapping.venues.derive {
    if derive_polled && !derive_suspect {
        if discovered_ids.contains(&(Venue::Derive, derive_inst.instrument.as_str())) {
            absence_tracker.record_present(Venue::Derive, &derive_inst.instrument);
        } else if absence_tracker.record_absent(Venue::Derive, &derive_inst.instrument) {
            tracing::warn!(

                instrument = %derive_inst.instrument,
                venue = "derive",
                "instrument absent for {} consecutive polls, marking expired",
                absence_tracker.threshold
            );
            events_to_expire.insert(mapping.id.clone());
        }
    }
}
```
