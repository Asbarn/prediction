/// Count of active, unapproved (pending) mappings.
pub fn pending_count(&self) -> usize {
    self.mappings
        .iter()
        .filter(|m| !m.approved && m.status == LifecycleStatus::Active)
        .count()
}

```rust
for candidate in &candidates_to_append {
    let venue_names: Vec<&str> = [
        candidate.venues.deribit.as_ref().map(|_| "deribit"),
        candidate.venues.polymarket.as_ref().map(|_| "polymarket"),
        candidate.venues.kalshi.as_ref().map(|_| "kalshi"),
    ]
    .into_iter()
    .flatten()
    .collect();

    tracing::warn!(
        event_id = %candidate.id,
        matched_venues = ?venue_names,
        deribit_instrument = ?candidate.venues.deribit.as_ref().map(|d| &d.instrument),
        polymarket_instrument = ?candidate.venues.polymarket.as_ref().map(|p| &p.condition_id),
        kalshi_instrument = ?candidate.venues.kalshi.as_ref().map(|k| &k.ticker),
        expiry = %candidate.expiry,
        confidence = %candidate.expiry_confidence,
        "new proposal: candidate mapping discovered"
    );
    metrics::counter!("proposals_total").increment(1);
}
```

This replaces the existing `tracing::info!` + `metrics::counter!("lifecycle_candidates_discovered")`. Keep `lifecycle_candidates_discovered` as well if desired (it tracks a different concept: total candidates found including deduplication), but the `proposals_total` counter specifically tracks new proposals written.
